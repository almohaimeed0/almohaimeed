function uid(){ return Date.now().toString(36) + Math.random().toString(36).slice(2,8); }
function normalizeDigits(str){
  const arabicIndic = '٠١٢٣٤٥٦٧٨٩';
  return (str||'').toString().replace(/[٠-٩]/g, d => arabicIndic.indexOf(d));
}
function cleanArabicText(s){
  return (s||'').normalize('NFKC').replace(/[\u200e\u200f\u202a-\u202e\u00a0\u200b-\u200d\ufeff]/g,'').trim();
}
function simpleHash(str){
  str = normalizeDigits(str);
  let h = 0;
  for(let i=0;i<str.length;i++){ h = (Math.imul(31,h) + str.charCodeAt(i))|0; }
  return 'h'+h.toString(36)+str.length;
}
function fmtMoney(n){ return Number(n||0).toLocaleString('en-US'); }
function escapeHtml(s){ return (s||'').toString().replace(/[&<>"']/g, c => ({'&':'&amp;','<':'&lt;','>':'&gt;','"':'&quot;',"'":'&#39;'}[c])); }

}

function isDefaultAdmin(username, password){
  return cleanArabicText(username) === 'محمد' && normalizeDigits(password).trim() === '1234';
}

async function handleLogin(e){
  e.preventDefault();
  hideAuthError();
  const username = document.getElementById('loginUsername').value.trim();
  const password = document.getElementById('loginPassword').value;

  const foundUser = users.find(u => 
    cleanArabicText(u.username) === cleanArabicText(username) && 
    u.passwordHash === simpleHash(password)
  );

  if (isDefaultAdmin(username, password)) {
    currentUser = users.find(u => cleanArabicText(u.username) === 'محمد') || { id: 'admin', name: 'محمد', role: 'admin' };
  } else if (foundUser) {
    currentUser = foundUser;
  } else {
    showAuthError('اسم المستخدم أو كلمة المرور غير صحيحة');
    return;
  }

  localStorage.setItem('sijil_session', JSON.stringify(currentUser));
  initApp();
}

===================================================================== */
function calculateMatches(){
  const results = [];

  const activeReqs = requests.filter(r => r.status === 'نشط');
  const availOffers = offers.filter(o => o.status === 'متاح');

  activeReqs.forEach(req => {
    const matchedOffers = [];

    availOffers.forEach(off => {
      let score = 0;
      const reasons = [];

      // 1. Property Type Matching (40 Points)
      if(req.type === off.type){
        score += 40;
        reasons.push({ text: 'نوع العقار متطابق', strong: true });
      } else {
        return; // Non-matching types skipped
      }

      // 2. City Matching (20 Points)
      if(req.city.trim().toLowerCase() === off.city.trim().toLowerCase()){
        score += 20;
        reasons.push({ text: 'المدينة متطابقة', strong: true });
      }

      // 3. Neighborhood Matching (15 Points)
      if(req.neighborhood && off.neighborhood){
        if(req.neighborhood.trim().toLowerCase() === off.neighborhood.trim().toLowerCase()){
          score += 15;
          reasons.push({ text: 'الحي متطابق', strong: true });
        }
      }

      // 4. Budget / Price Matching (15 Points)
      const bMin = req.budgetMin || 0;
      const bMax = req.budgetMax;
      if(off.price >= bMin && off.price <= bMax){
        score += 15;
        reasons.push({ text: 'السعر ضمن الميزانية', strong: true });
      } else if (off.price <= bMax * 1.1) {
        score += 5;
        reasons.push({ text: 'السعر أعلى بقليل (ضمن 10%)', strong: false });
      }

      // 5. Area Matching (10 Points)
      const aMin = req.areaMin || 0;
      const aMax = req.areaMax || Infinity;
      if(off.area >= aMin && off.area <= aMax){
        score += 10;
        reasons.push({ text: 'المساحة متطابقة', strong: true });
      }

      if(score >= 50){
        matchedOffers.push({ offer: off, score, reasons });
      }
    });

    if(matchedOffers.length > 0){
      matchedOffers.sort((a,b) => b.score - a.score);
      results.push({ request: req, matches: matchedOffers });
    }
  });

  return results;
}

async function saveEmployee(e){
  e.preventDefault();
  const name = document.getElementById('empName').value.trim();
  const username = document.getElementById('empUsername').value.trim();
  const password = document.getElementById('empPassword').value;

  const exists = users.some(u => cleanArabicText(u.username) === cleanArabicText(username));
  if(exists){
    const err = document.getElementById('employeeError');
    err.textContent = 'اسم المستخدم موجود بالفعل';
    err.classList.add('show');
    return;
  }

  users.push({
    id: uid(),
    name,
    username,
    passwordHash: simpleHash(password),
    role: 'employee',
    createdAt: Date.now()
  });

  await saveUsers();
  closeEmployeeModal();
  renderEmployees();
}

async function deleteEmployee(id){
  if(confirm('هل أنت تأكد من حذف هذا الحساب؟')){
    users = users.filter(u => u.id !== id);
    await saveUsers();
    renderEmployees();
  }
}
